```c
#include <stdio.h>  
#include <math.h>  
int Fibonacci(int i) {  
    if (i==1 || i==2)  
        return 1;  
    else  
        return Fibonacci(i-1) + Fibonacci(i-2);  
}  
int main() {  
    int n;  
    scanf("%d",&n);  
    if (n<40) {  
        printf("%d",Fibonacci(n));  
    }  
}
```
