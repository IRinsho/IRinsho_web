```c
#include <stdio.h>  
#include <math.h>  
int main() {  
    int year, month, day;  
    int day_of_month [13]={0,31,28,31,30,31,30,31,31,30,31,30,31};  
    while (scanf("%d-%d-%d", &year, &month, &day)!=EOF) {  
        int sum_of_days = 0;  
        if (year%4==0 && year%100!=0 || year%400==0) {  
            day_of_month[2] = 29;  
  
            for (int i=0; i<month; i++) {  
                sum_of_days += day_of_month[i];  
            }  
            sum_of_days += day;  
            int yu_of_day = 366-sum_of_days;  
            printf("%d", yu_of_day);  
        }  
        else if (year%4!=0 && year%100==0 || year%400!=0) {  
            for (int i=0; i<month; i++) {  
                sum_of_days += day_of_month[i];  
            }  
            sum_of_days += day;  
            int yu_of_day = 365-sum_of_days;  
            printf("%d", yu_of_day);  
        }  
    }  
}
```
