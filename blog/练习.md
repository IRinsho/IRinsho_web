```
#include <stdio.h>
#include <ctype.h>
int main() {
    char a;
    scanf("%c", &a);
    if (isalpha(a)) {
        if (isupper(a)) {
            a = tolower(a);
        }
        else if (islower(a)) {
            a = toupper(a);
        }
        printf("%c", a);
    }
    else
        printf("%c", a);


    return 0;
}
```
