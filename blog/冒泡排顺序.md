# 思路
两两比较相邻元素，如果前面的比后面大就交换。一轮下来最大的元素被放到最后
# 示例
```c
#include <stdio.h>  
#include <math.h>  
  
int main(void) {  
  
    int max;  
    int a[10];  
    for (int i=0;i<10;i++) {  
        scanf("%d",&a[i]);  
    }  
  
    max=a[0];  
    for (int t=0; t<10-1; t++) {  
        for (int j=0;j<10-1;j++) {  
             if (a[j]>a[j+1]) {  
                 max=a[j];  
                 a[j]=a[j+1];  
                 a[j+1]=max;  
            }  
        }  
    }  
    for (int i=0;i<10;i++) {  
        printf("%d\n",a[i]);  
    }  
  
return 0;  
  
}
```
